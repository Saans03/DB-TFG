DROP DATABASE IF EXISTS alojamiento;
CREATE DATABASE alojamiento;
USE alojamiento;

CREATE TABLE alojamiento (
    numaloj INT(2) PRIMARY KEY AUTO_INCREMENT,
    alojamiento CHAR(15),
    nombrecompleto CHAR(40),
    responsable CHAR(25),
    direccion CHAR(30),
    distancia INT
);
CREATE TABLE oficio (
    numoficio INT(2) PRIMARY KEY AUTO_INCREMENT,
    oficio CHAR(25),
    descripcion CHAR(80)
);
CREATE TABLE empleado (
    numemp INT(2) PRIMARY KEY AUTO_INCREMENT,
    nombre CHAR(25),
    edad INT,
    alojamiento INT(2),
    sueldo DOUBLE(5,3),
    FOREIGN KEY (alojamiento) REFERENCES alojamiento(numaloj) ON DELETE CASCADE
);
CREATE TABLE oficioempleado (
    empleado INT(2),
    oficio INT(2),
    calificacion CHAR(15),
    PRIMARY KEY (empleado, oficio),
    FOREIGN KEY (empleado) REFERENCES empleado(numemp) ON DELETE CASCADE,
    FOREIGN KEY (oficio) REFERENCES oficio(numoficio) ON DELETE CASCADE
);
